RED='\033[0;31m'; GREEN='\033[0;32m'; YELLOW='\033[1;33m'; BLUE='\033[0;34m'; CYAN='\033[0;36m'; NC='\033[0m'

header() {
    echo -e "\n${BLUE}════════════════════════════════════════════════════════════${NC}"
    echo -e "${BLUE} $1${NC}"
    echo -e "${BLUE}════════════════════════════════════════════════════════════${NC}"
    echo -e "\n--- $1 ---" >> "$REPORT_FILE"
}

fire_attack() {
    local source_container=$1
    local attack_desc=$2
    local command=$3
    local expected_sids=$4  # Puede aceptar formato "SID1|SID2"
    local wait_time=${5:-2} # Tiempo de espera por defecto: 2s (se sube para flood)

    local start_time=$(date '+%H:%M:%S')
    echo -e "${YELLOW}► [${start_time}] DESDE [${source_container}]:${NC} ${attack_desc}"

    # Ejecutamos el ataque en el contenedor correspondiente
    docker exec "$source_container" sh -c "$command" >/dev/null 2>&1 || true
    sleep "$wait_time"

    local end_time=$(date '+%H:%M:%S')

    # Buscamos coincidencias con cualquiera de los SIDs esperados usando Expresiones Regulares
    if docker exec ids_network_monitor sh -c "grep -E -q ':(${expected_sids}):' /var/log/snort/alert_json.txt 2>/dev/null"; then
        echo -e "${GREEN}   [✔] DETECTADO a las ${end_time} (SID interceptado: $expected_sids)${NC}"
        echo "[✔ DETECTADO] $attack_desc (SID objetivo: $expected_sids)" >> "$REPORT_FILE"
    else
        echo -e "${RED}   [✘] MITIGADO/CIEGO a las ${end_time} (Snort no vio el SID $expected_sids)${NC}"
        echo "[✘ BLOQUEO L3/L4 o FALLO] $attack_desc (SID objetivo: $expected_sids)" >> "$REPORT_FILE"
    fi
}

# Certificados mTLS estándar usados en el compose

# Limpiamos el entorno antes de la auditoría
